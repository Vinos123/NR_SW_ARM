# 中文：根工程汇总课程；各课也能从自己的目录独立配置。
# English: Aggregate lessons here; each also configures standalone.
foreach(number RANGE 1 24)
  if(number LESS 10)
    set(label "0${number}")
  else()
    set(label "${number}")
  endif()
  add_subdirectory(${label})
endforeach()
if(PUSCH_BUILD_EXERCISES)
  add_custom_target(all_exercises)
  foreach(number RANGE 1 24)
    if(number LESS 10)
      set(label "0${number}")
    else()
      set(label "${number}")
    endif()
    add_dependencies(all_exercises lesson_${label}_exercise)
  endforeach()
endif()
