cmake_minimum_required(VERSION 3.24)
project(pusch_lab VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PUSCH_BUILD_TESTS "Build reference tests" ON)
option(PUSCH_BUILD_EXERCISES "Build independent TODO exercises" ON)
option(PUSCH_ENABLE_AVX512 "Build separately dispatched AVX-512 kernel" OFF)
option(PUSCH_ENABLE_SVE "Build separately dispatched SVE kernel (Linux ARM)" OFF)

add_library(kissfft STATIC third_party/kissfft/kiss_fft.c)
target_include_directories(kissfft PUBLIC third_party/kissfft)
add_library(srsran_ldpc STATIC
  third_party/srsran/ldpc/ldpc_decoder_generic.cpp
  third_party/srsran/ldpc/ldpc_decoder_impl.cpp
  third_party/srsran/ldpc/ldpc_graph_impl.cpp
  third_party/srsran/ldpc/ldpc_luts_impl.cpp
  third_party/srsran/log_likelihood_ratio.cpp)
target_include_directories(srsran_ldpc SYSTEM PUBLIC
  third_party/srsran/include third_party/srsran/ldpc third_party/srsran/fmt)
target_compile_definitions(srsran_ldpc PUBLIC FMT_HEADER_ONLY)
add_library(pusch STATIC src/modulation.cc src/coding.cc src/decoder_adapter.cc
  src/waveform.cc src/link.cc)
target_include_directories(pusch PUBLIC include)
target_link_libraries(pusch PRIVATE kissfft srsran_ldpc)
target_compile_options(pusch PRIVATE -Wall -Wextra -Wpedantic)
if(PUSCH_BUILD_TESTS)
  enable_testing()
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/googletest EXCLUDE_FROM_ALL)
  add_executable(pusch_tests tests/core_test.cc tests/reference_test.cc)
  target_link_libraries(pusch_tests PRIVATE pusch GTest::gtest_main)
  target_compile_definitions(pusch_tests PRIVATE PUSCH_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  include(GoogleTest)
  gtest_discover_tests(pusch_tests)
endif()

target_sources(pusch PRIVATE src/simd.cc)
if(PUSCH_ENABLE_AVX512)
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
    message(FATAL_ERROR "AVX-512 requires x86-64")
  endif()
  add_library(pusch_avx512 OBJECT src/qpsk_avx512.cc)
  target_compile_options(pusch_avx512 PRIVATE -mavx512f)
  target_sources(pusch PRIVATE $<TARGET_OBJECTS:pusch_avx512>)
  target_compile_definitions(pusch PRIVATE PUSCH_HAS_AVX512)
endif()
if(PUSCH_ENABLE_SVE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    message(FATAL_ERROR "SVE requires ARM64 Linux")
  endif()
  add_library(pusch_sve OBJECT src/qpsk_sve.cc)
  target_compile_options(pusch_sve PRIVATE -march=armv8-a+sve)
  target_sources(pusch PRIVATE $<TARGET_OBJECTS:pusch_sve>)
  target_compile_definitions(pusch PRIVATE PUSCH_HAS_SVE)
endif()
add_executable(pusch_cli src/main.cc)
target_link_libraries(pusch_cli PRIVATE pusch)

add_subdirectory(lessons)
