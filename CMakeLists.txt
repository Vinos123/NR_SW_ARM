cmake_minimum_required(VERSION 3.24)
project(pusch_lab VERSION 1.1 LANGUAGES C CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PUSCH_BUILD_TESTS "Build reference tests" ON)
option(PUSCH_BUILD_EXERCISES "Build TODO exercise targets" ON)
option(PUSCH_ENABLE_AVX512 "Enable separately compiled AVX-512 kernels" OFF)
option(PUSCH_ENABLE_SVE "Enable separately compiled SVE kernels" OFF)
include(cmake/PuschModules.cmake)

# 中文：仅完整链路和总测试使用聚合接口，课程各自声明最小依赖。
# English: Only integration/tests use the aggregate; lessons declare modules.
add_library(pusch INTERFACE)
target_include_directories(pusch INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(pusch INTERFACE cxx_std_20)
foreach(module link equalization power_control simd)
  pusch_require_module(${module})
  target_link_libraries(pusch INTERFACE pusch_${module})
endforeach()
add_executable(pusch_cli src/main.cc)
target_link_libraries(pusch_cli PRIVATE pusch)

if(PUSCH_BUILD_TESTS)
  enable_testing()
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/googletest EXCLUDE_FROM_ALL)
  add_executable(pusch_tests tests/core_test.cc tests/reference_test.cc
    tests/portable_math_test.cc)
  target_link_libraries(pusch_tests PRIVATE pusch GTest::gtest_main)
  target_compile_definitions(pusch_tests PRIVATE
    PUSCH_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  include(GoogleTest)
  gtest_discover_tests(pusch_tests)
endif()
add_subdirectory(lessons)
